**C++的构造函数**

```c++
#include <iostream>

/*
1. 函数名与类名相同。
2. 无返回值。

4. 构造函数可以重载。

{

	Date()

		_year = 0;
		_month = 1;
		_day = 1;
	}

		_year = year;
```

# 3.析构函数

```
1. 析构函数名是在类名前加上字符 ~。
2. 无参数无返回值。
3. 一个类有且只有一个析构函数。若未显式定义，系统会自动生成默认的析构函数。
4. 对象生命周期结束时，C++编译系统系统自动调用析构函数。
关于编译器自动生成的析构函数，是否会完成一些事情呢？
编译器生成的默认析构函数，对会自定类型成员调用它的析构函数。
```

```c++
#include <iostream>
using namespace std;

class hour
{
public:
	hour(int hours = 1, int minutes = 1, int seconds = 1)
		:_hours(hours)
		,_minutes(minutes)
		,_seconds(seconds)
	{
		cout << "this is hour" << endl;
	}

	~hour()
	{
		cout << "~hour" << endl;
	}

private:
	int _hours;
	int _minutes;
	int _seconds;

};

class Date
{
public:
	void Display()
	{
		cout << _year << "-" << _month << "-" << _day << endl;
	}

    ~Date()
    {
		cout << "~Date" << endl;
    }
    
private:
	 int _year;
	 int _month;
	 int _day;
	 hour h;  // c++会调用自定义类型的默认构造函数
};

int main()
{
	Date d; // 编译器默认的构造函数
	d.Display();  // 我们不写，他会调用自定义类型的析构函数。
	return 0;
}
```

**构造函数和析构函数：编译器默认生成的会调用自定义类型的构造函数和析构函数。**

# 4. 拷贝构造函数

**让新的对象和已经存在的对象一模一样的。**

```c++
#include <iostream>
using namespace std;

class Date
{
public:
	Date(int year = 0, int month = 1, int day = 1)
	{
		_year = year;
		_month = month;
		_day = day;
	}

	~Date()
	{
		cout << "~Date" << this << endl;
	}

	// 注意拷贝构造会无穷递归下去的。
	// Date(Date d)  Date d = d1; 一直递归下去的
	Date(const Date& d)  // 这里const可以传递：const和非const对象
	{
		_year = d._year;
		_month = d._month;
		_day = d._day;
	}

	void Display()
	{
		cout << _year << "-" << _month << "-" << _day << endl;
	}

private:
	 int _year;
	 int _month;
	 int _day;

};

// 传参就是一个拷贝构造的过程：传递的参数构造一个d
void func(Date d)  // 拷贝构造 Date d = d2;
{}

int main()
{
	Date d1(1, 2, 3);
	d1.Display();

	Date d2(d1);  // 拷贝构造
	d2.Display();

	Date d3 = d2; // 拷贝构造，就是为什么会形成无穷递归的原因的(传递值的时候)
	d3.Display();

	func(d2); // 拷贝构造

	return 0;
}
```



# 5.赋值运算符重载

**返回值类型 operator操作符(参数列表)**

```c++
#include <iostream>
using namespace std;

class Date
{
public:
	Date(int year = 0, int month = 1, int day = 1)
	{
		_year = year;
		_month = month;
		_day = day;
	}

	~Date()
	{
		cout << "~Date" <<  endl;
	}

	// 注意拷贝构造会无穷递归下去的。
	// Date(Date d)  Date d = d1; 一直递归下去的
	Date(const Date& d)  // 这里const可以传递：const和非const对象
	{
		_year = d._year;
		_month = d._month;
		_day = d._day;
	}

	void Display()
	{
		cout << _year << "-" << _month << "-" << _day << endl;
	}

	bool operator==(const Date& d) const
	{
		return _year == d._year && _month == d._month && _day == d._day;
	}

	bool operator!=(const Date& d) const
	{
		// return !(*this == d);
		return !(*this).operator==(d);
	}

	bool operator>(const Date& d) const
	{
		return _year > d._year || 
			  (_year == d._year && _month > d._month) || 
			  (_year == d._day && _month == d._month && _day > d._day);
	}

	bool operator<(const Date& d) const
	{
		return d > *this;
	}

	bool operator>=(const Date& d) const
	{
		return *this > d || *this == d;
	}

	bool operator<=(const Date& d) const
	{
		return *this < d || *this == d;
	}

	Date& operator=(const Date& d)
	{
		_year = d._year;
		_month = d._month;
		_day = d._day;
	}

private:
	 int _year;
	 int _month;
	 int _day;

};
```

# 日期类的实现

```c++
#include <iostream>
using namespace std;

class Date
{
private:
	int GetMonthDay(int year, int month)
	{
		static const int _monthDays[13] = {0, 31, 28, 31, 30, 31, 30,31, 31, 30, 31, 30, 31};
		if (month == 2 && (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)))
		{
			return 29;
		}

		return _monthDays[month];
	}

public:
	Date(int year = 0, int month = 1, int day = 1)
	{
		if (year >= 0 && month >= 1 && month <= 12 && day >= 1 && day <= GetMonthDay(year, month))
		{
			_year = year;
			_month = month;
			_day = day;
		}
		else
		{
			cout << "非法日期" << endl;
		}
	}

	~Date()
	{
		cout << "~Date" <<  endl;
	}

	Date(const Date& d)  
	{
		_year = d._year;
		_month = d._month;
		_day = d._day;
	}

	Date& operator=(const Date& d)
	{
		_year = d._year;
		_month = d._month;
		_day = d._day;

		return *this;
	}

	void Display()
	{
		cout << _year << "-" << _month << "-" << _day << endl;
	}

	bool operator==(const Date& d) const
	{
		return _year == d._year && _month == d._month && _day == d._day;
	}

	bool operator!=(const Date& d) const
	{
		return !(*this == d);
	}

	bool operator>(const Date& d) const
	{
		return _year > d._year || 
			  (_year == d._year && _month > d._month) || 
			  (_year == d._year && _month == d._month && _day > d._day);
	}

	bool operator<(const Date& d) const
	{
		return d > *this;
	}

	bool operator>=(const Date& d) const
	{
		return !(*this < d);
	}

	bool operator<=(const Date& d) const
	{
		return  !(*this > d);
	}

	Date operator+(int day)
	{
		Date ret(*this);
		ret += day;
		return ret;
	}

	Date& operator+=(int day)
	{
		_day += day;
		while (_day > GetMonthDay(_year, _month))
		{
			_day -= GetMonthDay(_year, _month);
			++_month;

			if (_month == 13)
			{
				_year += 1;
				_month = 1;
			}
		}
		return *this;;
	}

	Date operator-(int day)
	{
		Date ret(*this);
		ret -= day;
		return ret;
	}

	Date& operator-=(int day)
	{
		_day -= day;

		while (_day <= 0)
		{
			--_month;
			if (_month == 0)
			{
				_month = 12;
				_year -= 1;
			}

			_day += GetMonthDay(_year, _month);
		}

		return *this;
	}

	Date& operator++()
	{
		*this += 1;
		return *this;
	}
	
	Date operator++(int)
	{
		Date ret(*this);
		++(*this);
		return ret;
	}
	
	Date& operator--()
	{
		*this -= 1;
		return *this;
	}
	
	Date operator--(int)
	{
		Date ret(*this);
		--(*this);
		return ret;
	}

private:
	 int _year;
	 int _month;
	 int _day;

};

int main()
{
	Date d1(1, 2, 4);
	Date d2(4, 5, 6);

	cout << (d1 > d2) << endl;
	cout << (d1 < d2) << endl;
	cout << (d1 >= d2) << endl;
	cout << (d1 <= d2) << endl;
	cout << (d1 == d2) << endl;
	cout << (d1 != d2) << endl;

	Date d3(2020, 4, 11);
	Date d4 = d3 + 10000;
	d4.Display();

	return 0;
}
```
