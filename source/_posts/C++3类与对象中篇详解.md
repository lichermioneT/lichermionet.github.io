# 2. 构造函数

**C++定义的时候就初始化了。 不需要再去调用成员函数了。**

```c++
#include <iostream>
using namespace std;

class Date
{
public:
	 void SetDate(int year, int month, int day)
	 {
		_year = year;
		_month = month;
		_day = day;
	 }

 void Display()
 {
	cout << _year << "-" << _month << "-" << _day << endl;
 }
private:
	 int _year;
	 int _month;
	 int _day;
};
int main()
{
	Date d1, d2;
	d1.SetDate(2018, 5, 1); // 忘了调用初始化呢？
	d1.Display();

	Date d2;
	d2.SetDate(2018, 7, 1); // 忘了调用初始化呢？
	d2.Display();
	return 0;
}
```

**C++的构造函数**

```c++
#include <iostream>
using namespace std;
/*
1. 函数名与类名相同。
2. 无返回值。
3. 对象实例化时编译器自动调用对应的构造函数。
4. 构造函数可以重载。
*/

class Date
{
public:
	Date()
	{
		_year = 0;
		_month = 1;
		_day = 1;
	}

	Date(int year, int month, int day)
	{
		_year = year;
		_month = month;
		_day = day;
	 }

	 void Display()
	 {
		cout << _year << "-" << _month << "-" << _day << endl;
	 }
private:
	 int _year;
	 int _month;
	 int _day;
};

int main()
{
	Date d(1, 3, 4); // 定义的时候调用，
	d.Display();

	Date d1; // 调用无参的构造函数 加括号变成函数的声明了。
	d1.Display();

	return 0;
}
```

```c++

	 int _year;
	 int _month;
	 int _day;

{
	Date d; // 编译器默认的构造函数
	d.Display();

	return 0;
}
```

**全缺省的默认构造函数**

```c++

// 一旦用户显式定义编译器将不再生成
class Date

public:
	Date(int year = 0, int month = 1, int day = 1)
	{
		_year = year;
		_month = month;
		_day = day;
	}

	void Display()
	{
		cout << _year << "-" << _month << "-" << _day << endl;
	}

private:

	 int _day;
};

int main()
{
	Date d; // 编译器默认的构造函数

```

**全缺省的构造函数-----最舒服的。  注意C++只能存在一个默认的构造函数。不需要传参的构造函数，就是默认的构造函数。**

# 3.析构函数

```
1. 析构函数名是在类名前加上字符 ~。
2. 无参数无返回值。
3. 一个类有且只有一个析构函数。若未显式定义，系统会自动生成默认的析构函数。
4. 对象生命周期结束时，C++编译系统系统自动调用析构函数。
关于编译器自动生成的析构函数，是否会完成一些事情呢？
编译器生成的默认析构函数，对会自定类型成员调用它的析构函数。
```

```c++
#include <iostream>
using namespace std;

class hour
{
public:
	hour(int hours = 1, int minutes = 1, int seconds = 1)
		:_hours(hours)
		,_minutes(minutes)
		,_seconds(seconds)
	{
		cout << "this is hour" << endl;
	}

	~hour()
	{
		cout << "~hour" << endl;
	}

private:
	int _hours;
	int _minutes;
	int _seconds;

};

class Date
{
public:

	void Display()
	{
		cout << _year << "-" << _month << "-" << _day << endl;
	}

private:
	 int _year;
	 int _month;
	 int _day;
	 hour h;  // c++会调用自定义类型的默认构造函数
};

int main()
{
	Date d; // 编译器默认的构造函数
	d.Display();  // 我们不写，他会调用自定义类型的析构函数。
	return 0;
}
```
