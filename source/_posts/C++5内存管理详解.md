# 1. C/C++内存分布

```c++
#include <cstdlib>

int globalVar = 1; // 数据区
static int staticGlobalVar = 1; // 数据区
void Test()
{
	static int staticVar = 1; // 数据区
	int localVar = 1;    // 栈区

	// num1栈区
	int num1[10] = { 1, 2, 3, 4 };

	// char2栈区 *char2栈区
	char char2[] = "abcd";

	// pChar3栈区 *pChar3代码段
	const char* pChar3 = "abcd";

	int* ptr1 = (int*)malloc(sizeof(int) * 4);
	int* ptr2 = (int*)calloc(4, sizeof(int));
	int* ptr3 = (int*)realloc(ptr2, sizeof(int) * 4);
	free(ptr1);
	free(ptr2);
	free(ptr3);
}
```
