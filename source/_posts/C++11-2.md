## 1.共享智能指针

### 1.1初始化

```c++
#include <iostream>
#include <memory>
using namespace std;

class test
{
public:
	test()
	{
		cout << "test()" << endl;
	}

	test(int x)
		:_num(x)
	{
		cout << "test(int x) : " << x << endl;
	}

	test(string str)
	{
		cout << "test(string str : " << str << endl;
	}

	~test()
	{
		cout << "~test()" << endl;
	}

private:
	int _num;
};

int main()
{
// 1.开辟堆内存初始化
	shared_ptr<int> ptr1(new int{ 44 });
	shared_ptr<int> ptr2(new int[5]{ 44 });

	cout << "ptr1的引用计数:" << ptr1.use_count() << endl; // 1
	cout << "ptr2的引用计数:" << ptr2.use_count() << endl; // 1

// 2.拷贝构造
	shared_ptr<int> ptr3 = ptr1;
	shared_ptr<int> ptr4(ptr2);
	cout << "ptr1的引用计数:" << ptr1.use_count() << endl; // 2
	cout << "ptr2的引用计数:" << ptr2.use_count() << endl; // 2
	cout << "ptr3的引用计数:" << ptr3.use_count() << endl; // 2
	cout << "ptr4的引用计数:" << ptr4.use_count() << endl; // 2

// 2.1移动构造(右值进行初始化)
	shared_ptr<int> ptr5 = move(ptr3);
	shared_ptr<int> ptr6(move(ptr4));
	cout << "ptr1的引用计数:" << ptr1.use_count() << endl; // 2
	cout << "ptr2的引用计数:" << ptr2.use_count() << endl; // 2
	cout << "ptr3的引用计数:" << ptr3.use_count() << endl; // 0
	cout << "ptr4的引用计数:" << ptr4.use_count() << endl; // 0
	cout << "ptr5的引用计数:" << ptr5.use_count() << endl; // 2
	cout << "ptr6的引用计数:" << ptr6.use_count() << endl; // 2

// 3.make_share 模板函数
	shared_ptr<int> ptr7 = make_shared<int>(55);
	shared_ptr<test> ptr8 = make_shared<test>(6);  // 只需要类型，然后构造函数需要的参数类型。
	shared_ptr<test> ptr9 = make_shared<test>("cccccccccc");
	shared_ptr<test> ptr10 = make_shared<test>();
	cout << "ptr10的引用计数:" << ptr10.use_count() << endl; // 1

// 4.reset方法。 reset()里面需要传入指针，或者new新的内存。
	ptr10.reset();
	cout << "ptr10的引用计数:" << ptr10.use_count() << endl; // 0

	ptr10.reset(new test("licher"));
	cout << "ptr10的引用计数:" << ptr10.use_count() << endl; // 1
    
	return 0;
}
```

### 1.3 指定删除器

**为什么需要删除器，因为数组类型无法全部进行析构。**

```c++
#include <iostream>
#include <memory>
using namespace std;

class test
{
public:
	test()
	{
		cout << "test()" << endl;
	}

	test(int x)
		:_num(x)
	{
		cout << "test(int x) : " << x << endl;
	}

	test(string str)
	{
		cout << "test(string str : " << str << endl;
	}

	void print()
	{
		cout << "hello test" << endl;
	}

	~test()
	{
		cout << "~test()" << endl;
	}

private:
	int _num;
};

int main()
{
	// 单个类型随便用
	//shared_ptr<test> ppp(new test(100), [](test* t){
	//	cout << "------------" << endl;
	//	delete t;
	//	});
	
// 数组类型需要自定义删除器函数
	shared_ptr<test> ppp(new test[5], [](test* t) {
		cout << "------------" << endl;
		delete[] t;
		});

	shared_ptr<test> ppp2(new test[5], [](test* t) {
		cout << "------------" << endl;
		delete[] t;
		});

// 默认删除函数
	shared_ptr<int> p1(new int[10] {0}, default_delete<int[]>());
	shared_ptr<test> p2(new test[10], default_delete<test[]>());

	return 0;
}
```



## 2.独享智能指针

### 2.1初始化

```c++
#include <iostream>
#include <memory>
using namespace std;

int main()
{
// 1.初始化
	unique_ptr<int> ptr1(new int{ 33 }); // 堆内存初始化

// 2.移动方式。将亡值，纯右值。
	unique_ptr<int> ptr2(move(ptr1));
	unique_ptr<int> ptr3 = move(ptr2);

// 3.reset方法
	ptr3.reset(); // 释放资源了
	ptr3.reset(new int{ 555 });  // 重新初始化资源
	ptr3.reset(new int{ 444 });  // 再一次重新初始化资源

// 原始指针
	int* p = ptr3.get();
	cout << *p << endl; // 输出444
	cout << *ptr3 << endl;

	return 0;
}
```

### 2.2指定删除器

```c++
#include <iostream>
#include <functional>
#include <memory>
using namespace std;

class test
{
public:
	test()
	{
		cout << "test()" << endl;
	}

	test(int x)
		:_num(x)
	{
		cout << "test(int x) : " << x << endl;
	}

	test(string str)
	{
		cout << "test(string str : " << str << endl;
	}

	void print()
	{
		cout << "hello test" << endl;
	}

	~test()
	{
		cout << "~test()" << endl;
	}

private:
	int _num;
};

int main()
{
	using ptrFunc = void(*)(test*);
	unique_ptr<test,ptrFunc> ptr(new test("cccc"), [](test* t) {
		 cout << "-----1" << endl;
		 delete t;
		});

	unique_ptr<test, function<void(test*)>> ptr2(new test("cccc"), [=](test* t) {
		cout << "-----2" << endl;
		delete t;
		});

// 独占指针，能够管理数组
	unique_ptr<test[]> ptr3(new test[3]);

	// 在c++11不支持管理数组类型的地址，C++11以后才支持的。
	shared_ptr<test[]> ptr3(new test[3]);

	return 0;
}
```

## 4 POD类型

```c++
#include <iostream>
using namespace std;

struct Base { static int a; };
struct Child : public Base { int b; }; // pod类型

struct Base1 {  int a; };
struct Child2 : public Base { static int b; }; // pod类型

struct Child3 : public Base, public Base1 { static int d; }; // pod类型
struct Child4 : public Base1 { int d; }; // 不是pod类型

int main()
{   
// 1平凡类型：默认的六大函数。
// 1.1平凡的构造和析构函数
// 1.2平凡的拷贝构造和移动构造函数
// 1.3平凡的拷贝赋值和移动赋值运算
// 1.4不包含虚函数和虚基类(多重继承的时候，虚继承。)

// 2标准布局
// 2.1所以的非静态成员有相同的权限。
// 2.2继承关系的时候，基类和派生类只能一个出现非静态成员。多重继承也只能一个出现非静态成员
// 2.3派生类的第一个类型不是基类的类型

    return 0;
}
```

## 5 default 和 delete

**default ：只能是对象的六大函数。**

```c++
#include <iostream>
using namespace std;

class A 
{
public:
	A() = default;   // 默认的构造函数        
	~A() = default;  // 默认的析构函数        

	A(const A&) = default;      // 默认的拷贝构造   
	A(A&&) noexcept = default;  // 默认的移动构造   

	A& operator=(const A&) = default;      // 默认的拷贝赋值（拷贝赋值运算符）
	A& operator=(A&&) noexcept = default;  // 默认的移动赋值（移动赋值运算符）
};

int main()
{
	return 0;
}
```



**delete : 删除默认函数**

```c++
#include <iostream>
using namespace std;

class A 
{
public:
	A() = default;   // 默认的构造函数        
	~A() = default;  // 默认的析构函数        

	A(const A&) = delete;      // 默认的拷贝构造   
	A(A&&) noexcept = delete;  // 默认的移动构造   

	A& operator=(const A&) = delete;      // 默认的拷贝赋值（拷贝赋值运算符）
	A& operator=(A&&) noexcept = delete;  // 默认的移动赋值（移动赋值运算符）

	void print(int num)
	{
		cout << "num : " << num << endl;
	}

	void print(char ch) = delete;
};

int main()
{
	A().print(22);
	//A().print('a');  防止隐士类型的转换

	return 0;
}
```



## 6 友元

**c++11 声明类的时候，可以减少class关键字。**

**友元可以使用别名了的。**

```c++
#include <iostream>
#include <string>
using namespace std;

class Tom;
using Honey = Tom;

class Jack
{
	friend Tom;
private:
	string name = "jack";
	void print()
	{
		cout << "my name is " << name << endl;
	}
};

class Lucy
{
	friend Honey;
private:
	string name = "lucy";
	void print()
	{
		cout << "my name is " << name << endl;
	}
};

class Tom
{
public:
	void print()
	{
		cout << j.name << " : " << l.name << endl;
		j.print();
		l.print();
	}

private:
	Jack j;
	Lucy l;
};

int main()
{
	Tom t;
	t.print();

	return 0;
}
```
