## 1.共享智能指针

### 1.1初始化

```c++
#include <iostream>
#include <memory>
using namespace std;

class test
{
public:
	test()
	{
		cout << "test()" << endl;
	}

	test(int x)
		:_num(x)
	{
		cout << "test(int x) : " << x << endl;
	}

	test(string str)
	{
		cout << "test(string str : " << str << endl;
	}

	~test()
	{
		cout << "~test()" << endl;
	}

private:
	int _num;
};

int main()
{
// 1.开辟堆内存初始化
	shared_ptr<int> ptr1(new int{ 44 });
	shared_ptr<int[]> ptr2(new int[5]{ 44 }); // 注意c++11不支持数组，无法全部进行析构的

	cout << "ptr1的引用计数:" << ptr1.use_count() << endl; // 1
	cout << "ptr2的引用计数:" << ptr2.use_count() << endl; // 1

// 2.拷贝构造
	shared_ptr<int> ptr3 = ptr1;
	shared_ptr<int[]> ptr4(ptr2);
	cout << "ptr1的引用计数:" << ptr1.use_count() << endl; // 2
	cout << "ptr2的引用计数:" << ptr2.use_count() << endl; // 2
	cout << "ptr3的引用计数:" << ptr3.use_count() << endl; // 2
	cout << "ptr4的引用计数:" << ptr4.use_count() << endl; // 2

// 2.1移动构造(右值进行初始化)
	shared_ptr<int> ptr5 = move(ptr3);
	shared_ptr<int[]> ptr6(move(ptr4));
	cout << "ptr1的引用计数:" << ptr1.use_count() << endl; // 2
	cout << "ptr2的引用计数:" << ptr2.use_count() << endl; // 2
	cout << "ptr3的引用计数:" << ptr3.use_count() << endl; // 0
	cout << "ptr4的引用计数:" << ptr4.use_count() << endl; // 0
	cout << "ptr5的引用计数:" << ptr5.use_count() << endl; // 2
	cout << "ptr6的引用计数:" << ptr6.use_count() << endl; // 2

// 3.make_share 模板函数
	shared_ptr<int> ptr7 = make_shared<int>(55);   
	shared_ptr<test> ptr8 = make_shared<test>(6);  // 只需要类型，然后构造函数需要的参数类型。
	shared_ptr<test> ptr9 = make_shared<test>("cccccccccc");
	shared_ptr<test> ptr10 = make_shared<test>();
	cout << "ptr10的引用计数:" << ptr10.use_count() << endl; // 1

// 4.reset方法。释放自己的所有权。可以管理新的内存。
	ptr10.reset();
	cout << "ptr10的引用计数:" << ptr10.use_count() << endl; // 0

	ptr10.reset(new test("licher"));
	cout << "ptr10的引用计数:" << ptr10.use_count() << endl; // 1
    
	return 0;
}
```

## 4 POD类型

**平凡类型：类的默认六大函数 + 没有虚函数 + 没有虚基类(虚继承的对象)。**

```c++

{
    // 1平凡类型：默认的六大函数。平凡的pod类型。
    // 1.1平凡的构造和析构函数。       (默认的)
    // 1.2平凡的拷贝构造和移动构造函数 (默认的)
    // 1.3平凡的拷贝赋值和移动赋值运算 (默认的)
    // 1.4不包含虚函数和虚基类(多重继承的时候，虚继承(virtual继承)。)
    return 0;
```
