```c++
struct Color
{
	static constexpr unsigned int ALPHA_MASK = 0b1111'1111'0000'0000'0000'0000'0000'0000;
	static constexpr unsigned int RED_MASK = 0b0000'0000'1111'1111'0000'0000'0000'0000;
	static constexpr unsigned int GREEN_MASK = 0b0000'0000'0000'0000'1111'1111'0000'0000;
	static constexpr unsigned int BLUE_MASK = 0b0000'0000'0000'0000'0000'0000'1111'1111;

	Color(unsigned char a, unsigned char r, unsigned char g, unsigned char b)
	{
		argb = (a << 24) | (r << 16) | (g << 8)  | b;
	}

	unsigned char getAlpha() const
	{
		return argb >> 24;
	}

	unsigned char getRed() const
	{
		return  static_cast<unsigned char>((argb & RED_MASK) >> 16);
	}

	unsigned char getGreen() const
	{
		return  static_cast<unsigned char>((argb & GREEN_MASK) >> 8);
	}

	unsigned char geBlue() const
	{
		return static_cast<unsigned char>(argb & BLUE_MASK);
	}

	void print() const
	{
		cout << "ARGB: 0x"
			<< hex
			<< setfill('0')
			<< setw(8)
			<< argb
			<< endl;
	}
	unsigned int argb;
};

int main()
{
	Color c(125,200,100,255);
	c.print();

	cout << (int)c.getAlpha() << endl;
	cout << (int)c.getRed() << endl;
```

```c++
{
	return value * 10000;
}

// 1km=1000;
constexpr long double operator"" _km(long double value)
{
	return value * 1000.0L;
}

constexpr char operator"" _upper(char c)
{
	if (c >= 'a' && c <= 'z')
	{
		return c - 'a' + 'A';
	}

	return c;
}

// 字符串添加括号
string operator"" _ins(const char* str, size_t size)
{
	return string( "(") +str + ")";
}

// 重复添加
string operator"" _repeat(const char* str, size_t size)
{
	string result;
	for (int i = 0; i < 3; ++i)
	{
		result += str;
	}

	return result;
}
```

**C++ 14修改**

```c++
#include <iostream>
using namespace std;
// c++11 只能有一条return
// c++14
/*
C++11 中，
    1.被constexpr修饰的函数返回值不允许为 void
    2.被constexpr修饰的构造函数必须为空，只能通过初始化列表初始化所有成员(初始化需放到初始化列表)
    3.被constexpr修饰的类成员函数会隐式成为const 函数
C++14中,
    1.被constexpr修饰的函数允许返回值为 void
    2.被constexpr修饰的构造函数可以有函数体，可以包含控制流语句(初始化需放到初始化列表)
    3.被constexpr修饰的类成员函数体内部可以修改类成员的值。
*/

constexpr int fact(int n)
{
    int result = 1;
    for (int i = 1; i <= n; ++i)
        result *= i;
    return result;
}

class point
{
public:
    constexpr point(int x, int y)
        : _x(x), _y(y)
    {
        if (_x > 100 || _y > 100)
        {
            _x = _y = 100;          // 改成员，不是形参
        }
    }

    constexpr void setx(int v) { _x = v; }
    constexpr void sety(int v) { _y = v; }   // 去掉 const，加 constexpr

    constexpr int getx() const { return _x; }
    constexpr int gety() const { return _y; }

    // 注意：validate 非 const，只能对非 const 对象调用
    constexpr void validate()
    {
        if (_x * _y > 10000)
        {
            _x = 100;
            _y = 200;
        }
    }

private:
    int _y;
    int _x;
};

constexpr point movepoint(point p, int x, int y)
{
    p.setx(p.getx() + x);
    p.sety(p.gety() + y);
    return p;
}

int main()
{
    // cout << fact(10) << endl;

    constexpr point p1(100, 200);   // _y=200>100 → 截断为 100
    point p2{ 1, 4 };
    constexpr point p3 = p1;
    constexpr point p4 = movepoint(p1, 2, 3);

    // p1.validate();  // 错误：p1 是 const 对象，validate 非 const
    p2.validate();     // OK

    return 0;
}
```
