# 1 二进制字面量

```c++
#include <iostream>
using namespace std;

int main()
{
// 字面量:右值。
	123;
	12.4f;
	134.56;
	'c';
	"lichermionex";
	false;
	true;
	nullptr;
	NULL;
	0b1100;

// 二进制字面量:零B 零b
	int num1 = 0B1010;   // 8421
	int num2 = 0b1010;   // 8421
	int num3 = 0b11010;   // 16 8421
	cout << num1 << endl;
	cout << num2 << endl;
	cout << num3 << endl;

	auto x1 = 0b10u;
	auto x2 = 0b10ul;
	auto x3 = 0b10ull;

	cout << "x1 : " << typeid(x1).name() << endl;
	cout << "x2 : " << typeid(x2).name() << endl;
	cout << "x3 : " << typeid(x3).name() << endl;

	int y1 = 0b111'1111; // 不能够最后面和0b后面，也不能够连续。
	int y2 = 0xff'ff;
	int y3 = 077'77;
	int y4 = 11'111'111;

	return 0;
}
```

**例子：权限**

```c++
#include <iostream>
using namespace std;

enum Permission
{
	Read = 0b001,
	Write = 0b010,
	Execute = 0b100,
};

int main()
{
// 权限检查
	int user1 = 0b111;
	int user2 = 0b101;
	int user3 = 0b001;

	if (user2 & Permission::Read)
	{
		cout << "user2:有读权限" << endl;
	}

	if (user2 & Permission::Write)
	{
		cout << "user2:有写权限" << endl;
	}

	if (user2 & Permission::Execute)
	{
		cout << "user2:有执行权限" << endl;
	}

	return 0;
}
```

**掩码**

```c++
#include <iostream>
#include <iomanip>
using namespace std;

struct Color
{
	static constexpr unsigned int ALPHA_MASK = 0b1111'1111'0000'0000'0000'0000'0000'0000;
	static constexpr unsigned int RED_MASK = 0b0000'0000'1111'1111'0000'0000'0000'0000;
	static constexpr unsigned int GREEN_MASK = 0b0000'0000'0000'0000'1111'1111'0000'0000;
	static constexpr unsigned int BLUE_MASK = 0b0000'0000'0000'0000'0000'0000'1111'1111;

	Color(unsigned char a, unsigned char r, unsigned char g, unsigned char b)
	{
		argb = (a << 24) | (r << 16) | (g << 8)  | b;
	}

	unsigned char getAlpha() const
	{
		return argb >> 24;
	}

	unsigned char getRed() const
	{
		return  static_cast<unsigned char>((argb & RED_MASK) >> 16);
	}

	unsigned char getGreen() const
	{
		return  static_cast<unsigned char>((argb & GREEN_MASK) >> 8);
	}

	unsigned char geBlue() const
	{
		return static_cast<unsigned char>(argb & BLUE_MASK);
	}

	void print() const
	{
		cout << "ARGB: 0x"
			<< hex
			<< setfill('0')
			<< setw(8)
			<< argb
			<< endl;
	}
	unsigned int argb;
};

int main()
{
	Color c(125,200,100,255);
	c.print();

	cout << (int)c.getAlpha() << endl;
	cout << (int)c.getRed() << endl;
	cout << (int)c.getGreen() << endl;
	cout << (int)c.geBlue() << endl;

	return 0;
}
```

# 2 自定义字面量

**单一类型**

```c++
#include <iostream>
#include <string>
using namespace std;

// 1.整型类型：unsigned long long
// 2.浮点类型：long double
// 3.字符类型：char
// 4.字符串类型：const char*, size_t;
/*
返回值 operator"" _xxx(){}
*/

// 50000 = 5_wan
constexpr unsigned long long operator"" _wan(unsigned long long value)
{
	return value * 10000;
}

// 1km=1000;
constexpr long double operator"" _km(long double value)
{
	return value * 1000.0L;
}

constexpr char operator"" _upper(char c)
{
	if (c >= 'a' && c <= 'z')
	{
		return c - 'a' + 'A';
	}

	return c;
}

// 字符串添加括号
string operator"" _ins(const char* str, size_t size)
{
	return string( "(") +str + ")";
}

// 重复添加
string operator"" _repeat(const char* str, size_t size)
{
	string result;
	for (int i = 0; i < 3; ++i)
	{
		result += str;
	}

	return result;
}

int main()
{
// 自定义字面量
	unsigned long long dist = 5_wan;
	cout << "dist:" << dist << endl;

	long double  dist2 = 3.5_km;
	cout << "dist2:" << dist2 << endl;

	char ch = 'a'_upper;
	cout << "ch:" << ch << endl;

	string str1 = "lic"_ins;
	string str2 = "lic"_repeat;

	cout << "str1:" << str1 << endl;
	cout << "str2:" << str2 << endl;

	return 0;
}
```

**自定义类型**

```c++
#include <iostream>
#include <string>
using namespace std;

// 1.整型类型：unsigned long long
// 2.浮点类型：long double
// 3.字符类型：char
// 4.字符串类型：const char*, size_t;
/*
返回值 operator"" _xxx(){}
*/

struct color
{
	uint8_t r, g, b;
	
	void print()
	{
		cout << "r:" << (int)r << endl;
		cout << "g:" << (int)g << endl;
		cout << "b:" << (int)b << endl;
	}
};

color operator"" _rgb(const char* str, size_t size)
{
	int r, g, b;
	sscanf(str, "%d,%d,%d", &r, &g, &b);
	return color{ (uint8_t)r,(uint8_t)g,(uint8_t)b };
}

color operator"" _rgb2(unsigned long long value)
{
	int r, g, b;
	return color{ (uint8_t)((value >> 16) & 0xFF),(uint8_t)((value >> 8) & 0xFF),(uint8_t)(value & 0xFF) };
}

int main()
{
	color c1 = "255,100,100"_rgb;
	c1.print();

	color c2 = 0xff6699_rgb2;
	c2.print();

	return 0;
}
```
