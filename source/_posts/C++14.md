**C++ 14修改**

```c++
#include <iostream>
using namespace std;
// c++11 只能有一条return
// c++14
/*
C++11 中，
    1.被constexpr修饰的函数返回值不允许为 void
    2.被constexpr修饰的构造函数必须为空，只能通过初始化列表初始化所有成员(初始化需放到初始化列表)
    3.被constexpr修饰的类成员函数会隐式成为const 函数
C++14中,
    1.被constexpr修饰的函数允许返回值为 void
    2.被constexpr修饰的构造函数可以有函数体，可以包含控制流语句(初始化需放到初始化列表)
    3.被constexpr修饰的类成员函数体内部可以修改类成员的值。
*/

constexpr int fact(int n)
{
    int result = 1;
    for (int i = 1; i <= n; ++i)
        result *= i;
    return result;
}

class point
{
public:
    constexpr point(int x, int y)
        : _x(x), _y(y)
    {
        if (_x > 100 || _y > 100)
        {
            _x = _y = 100;          // 改成员，不是形参
        }
    }

    constexpr void setx(int v) { _x = v; }
    constexpr void sety(int v) { _y = v; }   // 去掉 const，加 constexpr

    constexpr int getx() const { return _x; }
    constexpr int gety() const { return _y; }

    // 注意：validate 非 const，只能对非 const 对象调用
    constexpr void validate()
    {
        if (_x * _y > 10000)
        {
            _x = 100;
            _y = 200;
        }
    }

private:
    int _y;
    int _x;
};

constexpr point movepoint(point p, int x, int y)
{
    p.setx(p.getx() + x);
    p.sety(p.gety() + y);
    return p;
}

int main()
{
    // cout << fact(10) << endl;

    constexpr point p1(100, 200);   // _y=200>100 → 截断为 100
    point p2{ 1, 4 };
    constexpr point p3 = p1;
    constexpr point p4 = movepoint(p1, 2, 3);

    // p1.validate();  // 错误：p1 是 const 对象，validate 非 const
    p2.validate();     // OK
```

```c++
#include <iostream>
#include <memory>
#include <string>
using namespace std;

// 智能指针：防止内存泄漏
// 创建方式

class test
{
public:
    test(int id, const string& name)
        :_id(id)
        ,_name(name)
    { }

    //test(){}
    test() = default;

private:
    int _id;
    string _name;
};

template<class T, typename... Arg>
unique_ptr<T> myunique(Arg&& ...args)
{
    return unique_ptr<T>(new T(forward<Arg>(args)...));
}

int main()
{
    // c11
    unique_ptr<int> ptr1{ new int{223} };
    unique_ptr<test> ptr2{ new test{223, "lic"} };
    unique_ptr<test> ptr3{ new test(223, "lic") };
    unique_ptr<test> ptr4 = std::move(ptr3);  // 独占，只能够交换所有权限

    auto ptr5 = myunique<int>(44);
    auto ptr6 = myunique<test>(44, "ssss"s);

    // c14 
    auto ptr7 = make_unique<test>(44, "sss"s);
    auto ptr8 = make_unique<int>(44);

    // c14管理数组
    //数组
    // ()中的数字代表数组的元素个数
    // 模板参数类型必须是T[]，[]不能丟，通过 delete[]释放内存
    // 数组元素初始化，使用默认构造函数
    auto ptr9 = make_unique<int[]>(12);   // 数组，元素12个；
    auto ptr10 = make_unique<test[]>(22); // 数组，元素22个； 自定义一个无参数的构造函数

    return 0;
}
```
