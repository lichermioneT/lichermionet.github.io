# 1二进制字面量

```c++
#include <iostream>
using namespace std;

int main()
{
// 字面量:右值。
	123;
	12.4f;
	134.56;
	'c';
	"lichermionex";
	false;
	true;
	nullptr;
	NULL;
	0b1100;

// 二进制字面量:零B 零b
	int num1 = 0B1010;   // 8421
	int num2 = 0b1010;   // 8421
	int num3 = 0b11010;   // 16 8421
	cout << num1 << endl;
	cout << num2 << endl;
	cout << num3 << endl;

	auto x1 = 0b10u;
	auto x2 = 0b10ul;
	auto x3 = 0b10ull;

	cout << "x1 : " << typeid(x1).name() << endl;
	cout << "x2 : " << typeid(x2).name() << endl;
	cout << "x3 : " << typeid(x3).name() << endl;

	int y1 = 0b111'1111; // 不能够最后面和0b后面，也不能够连续。
	int y2 = 0xff'ff;
	int y3 = 077'77;
	int y4 = 11'111'111;

	return 0;
}
```
