# 1.单例模型

**饿汉模式：线程安全的。**

```c++
#include <iostream>
using namespace std;

class TaskQueue
{
public:
	//TaskQueue() = delete;
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;

	// 静态成员属性
    static TaskQueue* getInstance()  // 通过静态函数得到的。
	{
		return _taskQ;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

private:
     TaskQueue() = default;
	// TaskQueue(const TaskQueue& t) = default;
	// TaskQueue& operator=(const TaskQueue& t) = default;
	
	static TaskQueue* _taskQ;  // 静态属性属于类
};

// 饿汉模式
TaskQueue* TaskQueue::_taskQ = new TaskQueue; //这里是调用默认的无参构造函数 

int main()
{
	TaskQueue* t = TaskQueue::getInstance();
	t->print();

	return  0;
}
```

**c++的默认构造函数有三个，编译默认的，全缺省，没有参数的构造函数。只能够出现一个的。**



**懒汉模式 + 双重检查锁定**

```c++
#include <iostream>
#include <mutex>
using namespace std;

class TaskQueue
{
public:
	//TaskQueue() = delete;
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;

	static TaskQueue* getInstance()
	{
		if (_taskQ == nullptr)
		{
			_mutex.lock();
			if (_taskQ == nullptr)
			{
				_taskQ = new TaskQueue;
			}
			_mutex.unlock();
		}
		return _taskQ;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

private:
     TaskQueue() = default;
	// TaskQueue(const TaskQueue& t) = default;
	// TaskQueue& operator=(const TaskQueue& t) = default;
	
	static TaskQueue* _taskQ;
	static mutex _mutex;
};

// 懒汉模式，需要的时候才去创建
TaskQueue* TaskQueue::_taskQ = nullptr;
mutex TaskQueue::_mutex; 

int main()
{
	TaskQueue* t = TaskQueue::getInstance();
	t->print();
	return  0;
}
```



**原子变量**

```c++
#include <iostream>
#include <mutex>
#include <atomic>
using namespace std;

class TaskQueue
{
public:
	//TaskQueue() = delete;
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;

	static TaskQueue* getInstance()
	{
		TaskQueue* task = _taskQ.load();

		if (task == nullptr)
		{
			_mutex.lock();

			task = _taskQ.load();
			if (task == nullptr)
			{
				task = new TaskQueue;
				_taskQ.store(task);
			}

			_mutex.unlock();
		}

		return task;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

private:
     TaskQueue() = default;
	// TaskQueue(const TaskQueue& t) = default;
	// TaskQueue& operator=(const TaskQueue& t) = default;
	
	static mutex _mutex;
	static atomic<TaskQueue*> _taskQ;
};

// 懒汉模式，需要的时候才去创建

atomic<TaskQueue*> TaskQueue::_taskQ;
mutex TaskQueue::_mutex; 

int main()
{
	TaskQueue* t = TaskQueue::getInstance();
	t->print();
	return  0;
}
```



**静态局部对象, 编译器支持c++11**

```c++
#include <iostream>
#include <mutex>
#include <atomic>

{
public:
	//TaskQueue() = delete;

     TaskQueue() = default;

	TaskQueue* t = TaskQueue::getInstance();

	return  0;
```

**举个例子**

```c++
#include <iostream>
#include <queue>
#include <mutex>
#include <chrono>
#include <thread>
using namespace std;

class TaskQueue
{
public:
	//TaskQueue() = delete;
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;

	static TaskQueue* getInstance()
	{
		return _task;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

	// 判断队列是否为空
	bool isEmpty()
	{
		lock_guard<mutex> locker(_mutex);
		return _data.empty();
	}

	// 添加任务
	void addTask(int x)
	{
		lock_guard<mutex> locker(_mutex);
		_data.push(x);
	}

	// 删除任务
	bool deleteTask()
	{
		lock_guard<mutex> locker(_mutex);
		if (!_data.empty())
		{
			_data.pop();
			return true;
		}

		return false;
	}

	// 取任务，不删除任务
	int takeTask()
	{
		lock_guard<mutex> locker(_mutex);
		if (!_data.empty())
		{
			return _data.front();
		}

		return -1;
	}

private:
     TaskQueue() = default;
	 static TaskQueue* _task;
	 queue<int> _data;
	 mutex _mutex;
};

TaskQueue* TaskQueue::_task = new TaskQueue;

int main()
{
	TaskQueue* t = TaskQueue::getInstance();

// 1.生产者
	thread t1([=]() {
		for (int i = 0; i < 100; ++i)
		{
			t->addTask(i + 100);
			cout << "addtask : " << (i + 100) << " 线程id:" << this_thread::get_id() << endl;
			this_thread::sleep_for(50ms);
		}
		});

// 2.消费者
	thread t2([=]() {
		while(!t->isEmpty())
		{
			this_thread::sleep_for(1000ms);
			int ret = t->takeTask();
			cout << "taketask : " << ret << " 线程id:" << this_thread::get_id() << endl;
			t->deleteTask();
		}
		});

	t1.join();
	t2.join();

	return  0;
}
```
