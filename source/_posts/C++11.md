## 1.1字符串原始字面量

**为了解决字符的转义和续行。**

```c++
#include <iostream>
#include <string>
using namespace std;

// 原始字符串就是为了解决c/c++字符的转义和续行

// 原始字符串语法
// R"tag(字符串)tag"

int main()
{
	string str1 = R"(lichermionex\r\n\t111)";
	cout << str1 << endl;

	cout << "lichermionex\r\n\t111" << endl;

	string str2 = R"name(lichermionex)name";
	cout << str2 << endl;

	return 0;
}
```

## 1.3 初始化列表

```c++
#include <iostream>	
#include <string>
using namespace std;

class base
{
public:
	base(int num)
		:a(num)
	{ }

// 非静态属性
	int a = 2;
	double b = 2.3;
	char ch = 'c';

	// 初始化列表
	int a2{3};
	double b2{ 2.4 };
	char ch2{'c'};
	string str{ "licher" };

// 静态属性
	static int c1;
	static const double c2;
	static const char* c3;
	
	void print() const
	{
		cout << "a " << a << endl;
	}

// 静态属性，只能在类外初始化
int base::c1 = 22;
const double base::c2 = 22.5;
const char* base::c3 = "lic";

int main()
{
	base b{ 1 };
	b.print();
	return 0;
}
```



## 1.4 final 和 override

```c++
#include <iostream>
using namespace std;

// final:修饰虚函数。此虚函数不能够被重写了。 写在虚函数的后面
// final:修饰对象，此对象不能够被继承了。 写在类的后面

class base
{
public:
	virtual void test()
	{
		cout << "base" << endl;
	}
};

class child final : public base
{
public:
	void test() final override
	{
		cout << "child" << endl;
	}
};

int main()
{

	child c;
	c.test();

	return 0;
}
```

**final 修饰虚函数，此虚函数被继承下去后不能够被重写了。**

**final 修饰对象，此对象不能够被继承了。**

```c++
#include <iostream>
using namespace std;

// 虚函数一旦为虚，后续整个继承链中始终为虚；override 不会取消 virtual，
// 只有 final 可以禁止继续重写。

class base
{
public:
	virtual void test()
	{
		cout << "base" << endl;
	}
};

class child : public base
{
public:
	void test() override
	{
		cout << "child" << endl;
	}
};

int main()
{

	return 0;
}
```

**检查继承下来的虚函数是否被重写了。**



## 1.5右尖括号的对其

```c++
#include <iostream>
#include <map>
#include <string>
using namespace std;

template<class T>
class base
{
public:
	void print(const T& t)
	{
		auto it = t.begin();
		while (it != t.end())
		{
			cout << it->first << ":" << it->second << endl;
			++it;
		}
	}
};

int main()
{
	map<int, string> m;
	m.insert(make_pair(1, "one"));
	m.insert(make_pair(2, "two"));
	m.insert(make_pair(3, "three"));

	base<map<int, string>> c; // c11之前是 右移操作符
	c.print(m);

	return 0;
}
```



```c++
#include <iostream>
#include <map>
#include <string>
using namespace std;

// 模板的模型类型
template<class T = long, class U = int>
void print(T t = 'A', U u = 'B')
{
	cout << "t : " << t << endl;
	cout << "u : " << u << endl;
 }

int main()
{
// 可以自动推导，指定的实参推导
	print('a', 'b');
	print<char>('a', 'b');
	print<int, char>('a', 'b');
	print<char, int >('a', 'b');

// 使用默认的
	print();

	return 0;
}
```

**没有指定需要推导的。默认的参数不能进行推导。**

## 1.6静态断言

**动态断言**

```c++
#include <cassert>
using namespace std;

// 动态断言：运行期间判断

char* createArray(int size)
{
	assert(size > 0);
	char* array = new char[size];
	return array;
}

int main()
{
	char* buf = createArray(0);
	return 0;
}
```

**编译期间不会报错，运行期间会报错的。**



**静态断言**

```c++
#include <cassert>
using namespace std;

// 静态断言：编译期间判断.

int main()
{
	static_assert(sizeof(char*) == 8, "当前不是64位机器");
	return 0;
}
```

**编译期间进行常量表达式的检查。**



## 1.7 noexcept

**异常**

```c++
#include <iostream>
#include <string>
using namespace std;

struct MyExcept
{
	MyExcept(string str)
		:msg(str)
	{ }

	string msg;
};

//void func() throw(int, double, MyExcept) // 异常类型限定
//void func() throw() // 不允许抛出异常
//void func() noexcept // 不允许抛出异常
void func() 
{
	try
	{
		//throw 1;
		throw MyExcept("string exception !");
	}
	catch (int e) // 内置类型异常
	{
		cout << "throw value : " << e << endl;
	}
	catch (MyExcept e) // 自定义类型的异常
	{
		cout << "MyException : " << e.msg << endl;
	}
}

int main()
{
	func();
	cout << "hello world" << endl;
	return 0;
}
```

**noexcept**

```c++
#include <iostream>
#include <string>
using namespace std;

struct MyExcept
{
	MyExcept(string str)
		:msg(str)
	{ }

	string msg;
};

void func() noexcept(sizeof(char*) == 4) // 32抛出
{
	throw MyExcept("string exception !");
}

int main()
{
	try
	{
		func();
	}
	catch (MyExcept e)
	{
		cout << e.msg << endl;
	}

	return 0;
}
```

**noexcept不允许函数抛出异常的。 noexcept后面可以根常量表达式，用来控制函数是否抛出异常的。**
