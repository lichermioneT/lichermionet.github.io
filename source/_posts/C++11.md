## 1.4 final 和 override

```c++
#include <iostream>
using namespace std;

// final:修饰虚函数。此虚函数不能够被重写了。 写在虚函数的后面
// final:修饰对象，此对象不能够被继承了。 写在类的后面

class base
{
public:
	virtual void test()
	{
		cout << "base" << endl;
	}
};

class child final : public base
{
public:
	void test() final override
	{
		cout << "child" << endl;
	}
};

int main()
{

	child c;
	c.test();

	return 0;
}
```

**final 修饰虚函数，此虚函数被继承下去后不能够被重写了。**

**final 修饰对象，此对象不能够被继承了。**


## 2.2 decltype

**基本使用**

```c++
#include <iostream>
#include <vector>
#include <string>
#include <map>
using namespace std;

class Test
{
public:
    int a;
    string str;
};

// decltype (表达式) ：根据表达式推导出类型，不会计算表达式的

int main() 
{
    int a = 22;
    decltype(a) b = 99;    // int b
    decltype(a + 3.14) c = 55.3; // double c
    decltype(a + b * c) d = 1.12; // double d

    int num = 10;
    const int& numm2 = num;

    decltype(num) num3 = 55;  // int num3;
    decltype(numm2) num4 = num; // const int num4;

    decltype(Test::a) num6 = 555;  // int num6
    decltype(Test::str) str = "licher"; // string str

    return 0;
}
```

**decltype(表达式） 变量名 = 初始值 。**



**函数返回值的使用**

```c++
#include <iostream>
#include <vector>
#include <string>
#include <map>
using namespace std;

// decltype (表达式) ：根据表达式推导出类型，不会计算表达式的
// 函数返回值

int func_int(){}
int& func_int_r(){}
int&& func_int_rr(){}

const int func_const_int() {}
const int& func_const_int_r() {}
const int&& func_const_int_rr(){}

int main() 
{
    decltype(func_int()) a = 22; // int a;
    decltype(func_int_r()) b = a; // int& a;
    decltype(func_int_rr()) b = 33; // int&& a;
// 字面量：纯右值
 decltype(func_const_int()) c = 22; // 纯粹的右值，直接去掉const， 就是 int c = 22;
    decltype(func_const_int_r()) d = a; // const int& d = a;
    decltype(func_const_int_rr()) e = 22; // cons int&& e = 22;

}
```

**纯右值，字面量的时候就是一个常量。 去掉const**



**被（）包围的时候或者是一个左值， decltype推导的时候是引用， 不能去掉 const和volatile限定符。**

```c++
#include <iostream>
#include <vector>
#include <string>
#include <map>
using namespace std;

// decltype (表达式) ：根据表达式推导出类型，不会计算表达式的
// 左值被括号包围，const和volatile不能去掉

class Test
{
public:
    int a;
    string str;
    const static int b = 22;
};

int main() 
{
    const Test test;
    decltype(test.a) a = 22; // int a = 22;
    decltype(test.b) b = 22; // const int b = 22;

    // const int& a2 = a;
    decltype((test.a)) a2 = a; // 带上const因为定义的时候，是const对象。c

    int n = 0;
    int m = 0;
    decltype(n + m) c = 0; // 纯右值，字面量， int c = 0;

    decltype(n = n + m) d = n; // 左值 int& d = n;

    return 0;
}
```



**decltype的应用**

```c++
#include <iostream>
#include <vector>
#include <string>

#include <map>

// decltype (表达式) ：根据表达式推导出类型，不会计算表达式的
// 左值被括号包围，const和volatile不能去掉

// 泛型编程：遍历容器

template<class T>
class Container
{
public:
    void print(T& t)
    {
        for (_it = t.begin(); _it != t.end(); ++_it)
        {
            cout << "value:" << *_it << endl;
        }
    }

private:
    decltype(T().begin()) _it; // 得到迭代器的类型
};

int main() 
{
    list<int> ls{ 1, 2,4,5,6,7,8, };

    Container<list<int>> c;
    c.print(ls);

    Container<const list<int>> c1;
    c1.print(ls);

    return 0;
}
```
