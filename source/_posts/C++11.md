# 1.1字符串原始字面量

**为了解决字符的转义和续行。**

```c++
#include <iostream>
#include <string>
using namespace std;

// 原始字符串就是为了解决c/c++字符的转义和续行

// 原始字符串语法
// R"tag(字符串)tag"

int main()
{
	string str1 = R"(lichermionex\r\n\t111)";
	cout << str1 << endl;

	cout << "lichermionex\r\n\t111" << endl;

	string str2 = R"name(lichermionex)name";
	cout << str2 << endl;

	return 0;
}
```



## 1.2 Long Long

```c++
#include <iostream>
using namespace std;

// long long c++11引入的

int main()
{
	// 1有符号
	long long num1 = 100LL;
	//long long int 
	//signed long long 
	//signed long long int

	// 2无符号
	unsigned long long num2 = 100ULL; // 建议大写
	unsigned long long int num22 = 100ULL; // 建议大写

	//内存大小必须最少八个字节
	cout << "long long max:" << LLONG_MAX << endl;
	cout << "long long min:" << LLONG_MIN << endl;
	cout << "unsigned long long max:" << ULLONG_MAX << endl;

	// char short int long longlong
	// unsigned: char short int long longlong

// 低等级到高等级的转换
	char c = 6;
	long long d = c;

// 低等级到高等级的转换

	unsigned long long aa = -1; // 有符号到无符号的
	cout << aa << endl;

	return 0;
}
```
