## 1.1字符串原始字面量

**为了解决字符的转义和续行。**

**语法："tag()tag" **

```c++
#include <iostream>
#include <string>
using namespace std;

// 原始字符串就是为了解决c/c++字符的转义和续行

// 原始字符串语法
// R"tag(字符串)tag"

int main()
{
	string str1 = "lichermionex\r\n\t111";
	string str2 = R"(lichermionex\r\n\t111)";
	string str3 = R"name(lichermionex\r\n\t111)name";

	cout << "str1 : " << str1 << endl;
	cout << "str1 : " << str2 << endl;
	cout << "str1 : " << str3 << endl;

	return 0;
}
```

**运行结果**

```

        111
str1 : lichermionex\r\n\t111
str1 : lichermionex\r\n\t111
```

## 1.4 final 和 override

```c++
#include <iostream>
using namespace std;

// final:修饰虚函数。此虚函数不能够被重写了。 写在虚函数的后面
// final:修饰对象，此对象不能够被继承了。 写在类的后面

class base
{
public:
	virtual void test()
	{
		cout << "base" << endl;
	}
};

class child final : public base  // 修饰对象，此对象不能够被继承下去了的
{
public:
	void test() final override  // 此虚函数不能够被从写了。
	{
		cout << "child" << endl;
	}
};

int main()
{

	child c;
	c.test();

	return 0;
}
```

**final 修饰虚函数，此虚函数被继承下去后不能够被重写了。**

**final 修饰对象，此对象不能够被继承了。**

```c++
#include <iostream>
using namespace std;

// 虚函数一旦为虚，后续整个继承链中始终为虚；override 不会取消 virtual，
// 只有 final 可以禁止继续重写。

class base
{
public:
	virtual void test()
	{
		cout << "base" << endl;
	}
};

class child : public base
{
public:
	void test() override  // 检查虚函数是否被重写成功了的
	{
		cout << "child" << endl;
	}
};

int main()
{
	return 0;
}
```

**检查继承下来的虚函数是否被重写了。**

## 2.2 decltype

```c++
#include <iostream>

{
    int a = 22;
    decltype(a) b = 99;    // int b
    decltype(a + 3.14) c = 55.3; // double c
    decltype(a + b * c) d = 1.12; // double d

    int num = 10;
    const int& numm2 = num;

    decltype(num) num3 = 55;  // int num3;
    decltype(numm2) num4 = num; // const int num4;

    decltype(Test::str) str = "licher"; // string str

}
```

**decltype(表达式） 变量名 = 初始值 。**



**函数返回值的使用**

```c++
 decltype(func_const_int()) c = 22; // 纯粹的右值，直接去掉const， 就是 int c = 22;
```

**被（）包围的时候或者是一个左值， decltype推导的时候是引用， 不能去掉 const和volatile限定符。**

```c++

#include <vector>
#include <string>
#include <map>

// 左值被括号包围，const和volatile不能去掉

class Test
{
public:
    int a;
    string str;
    const static int b = 22;
};

int main() 
{
    const Test test;
    decltype(test.a) a = 22; // int a = 22;
    decltype(test.b) b = 22; // const int b = 22;

    // const int& a2 = a;
    decltype((test.a)) a2 = a; // 带上const因为定义的时候，是const对象。c

    int n = 0;
    int m = 0;
    decltype(n + m) c = 0; // 纯右值，字面量， int c = 0;

    decltype(n = n + m) d = n; // 左值 int& d = n;

    return 0;
}
```

```c++
	// 成员函数绑定,成员函数属于对象，需要创建对象的示例。
	Test t;
	bind(&Test::output, &t, 1, 2)();
	bind(&Test::output, &t, 1, placeholders::_2)(777,888);

	// 类的成员变量,
	auto f = bind(&Test::_number, &t);

	f() = 777;
	cout << f() << endl;

	return 0;
}
```

**类的属性和方法属于实例化的对象，需要创建实例化的对象后 在继续进行绑定。**

## 3.8 右值引用

**左值：loactor value。可以定位的值**

**右值：read value。只读值**

**左值：空间**

**右值：数据**

```c++
#include <iostream>

// lvalue:loactor value,左值:定位值
// rvalue:read value    有值:只读值

// 左值引用，右值引用

int main()
{
// 左值
	int a = 44;

// 左值引用
	int& refa = a;

// 右值
	44;
	'a';
	"abcd";
    
// 右值引用
	int&& c = 44;
    
// 常量左值引用
	const int& d = a;
	const int& d2 = c;  // 常量右值引用
	const int& d3 = 44;  // 这里会有隐式类型转换
    
// 常量右值引用
	const int&& f = 44;

	return 0;
}
```

```c++
#include <iostream>
using namespace std;

class test
{
public:
	test()
		:_a(new int{ 22 })
	{
		cout << "test()" << endl;
		cout << &_a << endl;
	}

	// 拷贝构造：防止浅拷贝，实现深拷贝
	test(const test& t)
		:_a(new int{ *t._a })
	{
		cout << &_a << endl;
		cout << "test(const test& t)" << endl;
	}

	// 移动构造就是为了实现浅拷贝(有指针存在的话)。
	// 复用这个临时对象的资源。
	test(test&& t)
		:_a(t._a)
	{
		t._a = nullptr;
		cout << " test(test&& t)" << endl;
		cout << &_a << endl;
	}

	~test()
	{
		cout << "~test()" << endl;
		delete _a;
	}

public:
	int* _a;
};

test getobj()
{
	test t;  // 这里构造函数一次
	return t;
}

int main()
{
	// 右值引用：延长右值的生命周期。
	// 得到一个临时对象，
	test t = getobj(); // 这里一般拷贝构造一次。 但是存在移动构造，会调用拷贝构造。
	test&& t1 = getobj(); // 临时对象，才会进行移动构造的调用

	return 0;
}
```

**主要是延长将亡值的声明周期，实现浅拷贝。**

```c++
#include <iostream>
#include <string>
```

```c++
int main()
{
	int x = 1;
	int y = 2;

	auto&& v1 = x; // 左值
	auto&& v2 = 222; // 右值

	//decltype(x) && v3 = y; // error 右值引用赋了左值

	int&& a1 = 44;
	auto&& bb = a1; // 左值引用 int& b = a;
```

**通过右值推导 T&& 或者 auto&& 得到的是一个右值引用类型。 右值是具体的右值，不是变量。**
