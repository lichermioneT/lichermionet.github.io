# 1.继承的概念及定义

```c++
#include <iostream>
#include <string>
using namespace std;

class person
{
public:
	person(const string& name, int age, char sex)
		:_name(name)
		,_age(age)
		,_sex(sex)
	{ }

	void print()
	{
		cout << "姓名" << _name << endl;
		cout << "年龄" << _age << endl;
		cout << "性别" << _sex << endl;
	}

protected:
	string _name;
	int _age;
	char _sex;
};

class teacher : public person
{
public:
	teacher(const string& name, int age, char sex, int jobid)
		:person(name, age, sex)
		,_jobid(jobid)
	{ }

protected:
	int _jobid;
};

int main()
{
	string str("lic");
	teacher t(str, 1,'n',2);
	t.print();

	return 0;
}
```

# 2.基类和派生类对象赋值转换

```c++
#include <iostream>
#include <string>
using namespace std;

class person
{
public:
	person(const string& name, int age, char sex)
		:_name(name)
		,_age(age)
		,_sex(sex)
	{ }

	void print()
	{
		cout << "姓名" << _name << endl;
		cout << "年龄" << _age << endl;
		cout << "性别" << _sex << endl;
	}

protected:
	string _name;
	int _age;
	char _sex;
};

class teacher : public person
{
public:
	teacher(const string& name, int age, char sex, int jobid)
		:person(name, age, sex)
		,_jobid(jobid)
	{ }

protected:
	int _jobid;
};

int main()
{
	string str1("lic1");
	string str2("lic2");
	int age1 = 18;
	int age2 = 19;
	char ch1 = '1';
	char ch2 = '2';

	int jobid = 1;

	person p(str1, age1, ch1);
	teacher t(str2, age2, ch2, jobid);

	// 1.赋值
	p = t;
	p.print();
	
	// 2.指针
	person* ptr = &t;
	ptr->print();

	// 3.引用
	person& ref = t;
	ref.print();
	
	return 0;
}
```

**基类转换给派生类，需要强制类型转换。**

# 3.继承中的作用域

**派生类和基类的成员属性或者方法一样，就会形成隐藏的，或者叫做重定义的。**

**函数名一样就是 隐藏或者重定义的。不在乎函数的参数。**

```c++
#include <iostream>
#include <string>
using namespace std;

class person
{
public:
	person(const string& name, int age, char sex)
		:_name(name)
		,_age(age)
		,_sex(sex)
	{ }

	void print()
	{
		cout << "姓名" << _name << endl;
		cout << "年龄" << _age << endl;
		cout << "性别" << _sex << endl;
	}

protected:
	string _name;
	int _age;
	char _sex;
};

class teacher : public person
{
public:
	teacher(const string& name, int age, char sex, int jobid)
		:person(name, age, sex)
		,_jobid(jobid)
	{ }

	void pinrt()
	{
		person::print();  // // 指定作用域
		cout << "老师的id" << _jobid << endl;
	}

protected:
	int _jobid;
};

int main()
{
	string str1("lic1");
	string str2("lic2");
	int age1 = 18;
	int age2 = 19;
	char ch1 = '1';
	char ch2 = '2';

	int jobid = 1;

	person p(str1, age1, ch1);
	teacher t(str2, age2, ch2, jobid);

	t.person::print(); // 指定作用域

	return 0;
}
```

**1.重载：必须同一个作用域。**

**2.重写：多态。**

**3.重定义：也叫做隐藏。**

# 4.派生类的默认成员函数

```c++
#include <iostream>
#include <string>
using namespace std;

class person
{
public:
	person(const string& name, int age, char sex)
		:_name(name)
		,_age(age)
		,_sex(sex)
	{ 
		cout << "构造函数：person(const string& name, int age, char sex)" << endl;
	}

	person(const person& p)
	{
		_name = p._name;
		_age = p._age;
		_sex = p._sex;

		cout << "拷贝构造 person(const person& p)" << endl;
	}

	person& operator=(const person& p)
	{
		_name = p._name;
		_age = p._age;
		_sex = p._sex;

		cout << "赋值构造 person(const person& p)" << endl;

		return *this;
	}

	~person()
	{
		cout << "person的析构函数" << endl;
	}

	void print()
	{
		cout << "姓名" << _name << endl;
		cout << "年龄" << _age << endl;
		cout << "性别" << _sex << endl;
	}


protected:
	string _name;
	int _age;
	char _sex;
};

class teacher : public person
{
public:
	teacher(const string& name, int age, char sex, int jobid)
		:person(name, age, sex)
		,_jobid(jobid)
	{ }

	teacher(const teacher& t)
		: person(t)          // 调用父类的拷贝构造，复制父类成员
		, _jobid(t._jobid)    // 复制子类自己的成员
	{
		cout << "拷贝构造 teacher(const teacher& t)" << endl;
	}

	teacher& operator=(const teacher& t)
	{
		if (this != &t)
		{
			person::operator=(t);  // 给父类部分赋值
			_jobid = t._jobid;     // 给子类自己的成员赋值
		}

		return *this;
	}

	~teacher()
	{
		cout << "老师的析构函数" << endl;
	}

	void pinrt()
	{
		person::print();
		cout << "老师的id" << _jobid << endl;
	}

protected:
	int _jobid;
};

int main()
{
	string str1("lic1");
	string str2("lic2");
	int age1 = 18;
	int age2 = 19;
	char ch1 = '1';
	char ch2 = '2';

	int jobid = 1;

	teacher t(str2, age2, ch2, jobid);
	teacher t2(t);

	return 0;
}
```
